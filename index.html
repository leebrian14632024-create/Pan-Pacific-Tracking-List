// 修正後的各大船公司官方即時查詢直達網址 function getCarrierTrackingUrl(carrier, mbl, cntr) { const queryNumber = (mbl || cntr || "").split('/')[0].trim(); if (!queryNumber) return "#"; switch(carrier) { case "Evergreen": // 長榮 ShipmentLink Quick Tracking 直達 return `https://www.shipmentlink.com/servlet/TTrk_Tracking?bk_no=${queryNumber}&type=BL`; case "ONE": // ONE Ocean Network Express 官方查詢頁 return `https://ecomm.one-line.com/one-ecom/manage-shipment/cargo-tracking?trackingType=BL&trackingNumber=${queryNumber}`; case "Yang Ming": // 陽明海運最新貨況追蹤直達頁 return `https://www.yangming.com/e-service/track_trace/track_trace_cargo_tracking.aspx?type=bl&num=${queryNumber}`; case "Wan Hai": // 萬海航運 Cargo Tracking 直達頁 return `https://www.wanhai.com/views/cargoTracking/cargoTracking.xhtml?blNo=${queryNumber}`; case "ZIM": // 以星綜合航運直達追蹤 return `https://www.zim.com/tools/track-a-shipment?consignmentNumber=${queryNumber}`; case "Hapag-Lloyd": // 赫伯羅特 Online Business Tracking return `https://www.hapag-lloyd.com/en/online-business/track/track-by-booking-solution.html?blno=${queryNumber}`; default: return `https://www.google.com/search?q=${encodeURIComponent(carrier + ' tracking ' + queryNumber)}`; } }
